========== 4. Job di baseline ==========
[fbellina2@wilson es03]$ cat baseline.sh
#!/usr/bin/bash
#SBATCH --job-name=es03_baseline
#SBATCH --output=%x.%j.out
#SBATCH --error=%x.%j.err
#SBATCH --nodes=1
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=16
#SBATCH --hint=nomultithread
#SBATCH --exclusive
#SBATCH --mem=0
#SBATCH --partition=wilson
#SBATCH --constraint=sandy_bridge
#SBATCH --time=01:00:00

set -uo pipefail
cd "$SLURM_SUBMIT_DIR"
module load GCC/13.2.0
echo "Job: $SLURM_JOB_ID; nodo: $SLURM_JOB_NODELIST"
lscpu
module list 2>&1
command -v perf && perf --version

one() { srun --ntasks=1 --cpus-per-task=1 --cpu-bind=cores "$@"; }

echo "=== 1. Baseline seriale: 256 token, -t 0, prompt \"Once upon a time\""
for m in 15M 42M 110M; do
    for r in 1 2 3; do
        echo "--- stories$m run $r"
        one ./run "/srv/data/wilson/project/hpc4ai/exercises/tiny_stories/stories$m.bin" -t 0 -n 256 -i "Once upon a time" > "out_${m}_r$r.txt" 2> "tok_${m}_r$r.txt"
        cat "tok_${m}_r$r.txt"
    done
    cmp -s "out_${m}_r1.txt" "out_${m}_r2.txt" && cmp -s "out_${m}_r1.txt" "out_${m}_r3.txt" \
        && echo "stories$m: testo identico nelle 3 ripetizioni" || echo "stories$m: TESTO DIVERSO fra le ripetizioni"
done

echo "=== 2. Profilo con gprof"
for m in 15M 110M; do
    mkdir -p "gprof_$m" && ln -sf ../tokenizer.bin "gprof_$m/" && ( cd "gprof_$m" && one ../run_pg "/srv/data/wilson/project/hpc4ai/exercises/tiny_stories/stories$m.bin" -t 0 -n 256 -i "Once upon a time" > /dev/null 2>&1 \
        && gprof -b -p ../run_pg gmon.out | head -25 )
done
if command -v perf >/dev/null; then
    echo "=== 2b. Profilo con perf (110M)"
    one perf record -o perf_110M.data ./run "/srv/data/wilson/project/hpc4ai/exercises/tiny_stories/stories110M.bin" -t 0 -n 256 -i "Once upon a time" > /dev/null 2>&1
    perf report -i perf_110M.data --stdio --sort symbol 2>/dev/null | grep -v '^#' | grep -v '^$' | head -20
fi

echo "=== 3. Banda di memoria (triad)"
for t in 1 2 4 8 16; do
    OMP_NUM_THREADS=$t OMP_PLACES=cores OMP_PROC_BIND=spread \
        srun --ntasks=1 --cpus-per-task=16 --cpu-bind=cores ./triad
done
